1   package fr.univtln.bruno.samples.java101.tp2.impl;
2   
3   import fr.univtln.bruno.samples.java101.tp2.api.Drivable;
4   import fr.univtln.bruno.samples.java101.tp2.api.Refuelable;
5   import fr.univtln.bruno.samples.java101.tp2.model.MotorVehicle;
6   
7   /**
8    * Concrete Vehicle implementation representing a car.
9    * Designed to stay small and focused: minimal state and behaviour.
10   */
11  public class Car extends MotorVehicle implements Drivable, Refuelable {
12  
13      private double fuelLevel; // percentage 0.0 .. 100.0
14  
15      public Car(String id) {
16          super(id);
17          this.fuelLevel = 100.0; // default full tank for demo
18      }
19  
20      public Car(String id, String make, String model, int year) {
21          super(id, make, model, year);
22          this.fuelLevel = 100.0;
23      }
24  
25      /** {@inheritDoc} */
26      @Override
27      public String move() {
28          if (fuelLevel <= 0) return "cannot move (no fuel)";
29          // simulate fuel consumption
30          fuelLevel = Math.max(0.0, fuelLevel - 5.0);
31          return "drives";
32      }
33  
34      /** {@inheritDoc} */
35      @Override
36      public String drive() {
37          return move();
38      }
39  
40      /**
41       * Returns the current fuel level as a percentage.
42       * @return fuel level in percent.
43       */
44      public double getFuelLevel() {
45          return fuelLevel;
46      }
47  
48      /**
49       * Refuel the car to the specified percentage (clamped 0..100).
50       * @param percent new fuel level
51       */
52      public void refuel(double percent) {
53          this.fuelLevel = Math.max(0.0, Math.min(100.0, percent));
54      }
55  
56      /**
57       * Provides the default Drivable start behaviour for cars.
58       * @return the Drivable default starting label.
59       */
60      @Override
61      public String start() {
62          if (fuelLevel <= 0) return "cannot start (no fuel)";
63          return Drivable.super.start();
64      }
65  
66      /**
67       * Static category used to demonstrate method hiding.
68       * @return the category label for cars.
69       */
70      public static String category() {
71          return "Car";
72      }
73  
74      /**
75       * Covariant copy: returns a Car instance.
76       * @return a new Car with the same identifier.
77       */
78      @Override
79      public Car copy() {
80          Car c = new Car(this.id, this.make, this.model, this.year);
81          c.fuelLevel = this.fuelLevel;
82          return c;
83      }
84  
85      /** {@inheritDoc} */
86      @Override
87      public String getPowerSource() {
88          return "Internal combustion engine (fuel)";
89      }
90  }
