3   import fr.univtln.bruno.samples.java101.tp2.api.Drivable;
4   
5   /**
6    * Simple service object illustrating delegation: the driver delegates
7    * driving operations to a Drivable vehicle it holds.
8    */
9   public class Driver {
10      private final Drivable vehicle;
11      private final String name;
12  
13      public Driver(String name, Drivable vehicle) {
14          this.name = name;
15          this.vehicle = vehicle;
16      }
17  
18      /**
19       * Delegates the drive call to the underlying Drivable vehicle.
20       * @return the action verb from the vehicle's implementation.
21       */
22      public String drive() {
23          return vehicle.drive();
24      }
25  
26      /**
27       * Delegates the start call to the underlying Drivable vehicle.
28       * @return the default or overridden start label from the vehicle.
29       */
30      public String startVehicle() {
31          return vehicle.start();
32      }
33  
34      /**
35       * @return the driver's name.
36       */
37      public String getName() {
38          return name;
39      }
40  }
