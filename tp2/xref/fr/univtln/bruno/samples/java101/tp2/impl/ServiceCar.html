5   /**
6    * Service car combining Car's drivable features with maintenance checks.
7    * Demonstrates resolving default-method conflicts by explicitly selecting
8    * both sequences and composing them.
9    */
10  public class ServiceCar extends Car implements Maintainable {
11  
12      public ServiceCar(String id) {
13          super(id);
14      }
15  
16      /**
17       * Resolves the default-method conflict by composing Maintainable and Car
18       * behaviours in a defined order.
19       * @return the composed start sequence description.
20       */
21      @Override
22      public String start() {
23          // call Maintainable default then Car's implementation (which delegates to Drivable)
24          String m = Maintainable.super.start();
25          String d = super.start();
26          return m + " + " + d;
27      }
28  
29      /**
30       * Covariant copy: returns a ServiceCar instance.
31       * @return a new ServiceCar with the same identifier.
32       */
33      @Override
34      public ServiceCar copy() {
35          return new ServiceCar(getId());
36      }
37  }
