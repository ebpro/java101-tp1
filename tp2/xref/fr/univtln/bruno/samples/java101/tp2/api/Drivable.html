3   /**
4    * Drivable capability contract.
5    * Demonstrates multiple inheritance via interfaces and a default method.
6    */
7   public interface Drivable {
8       /**
9        * Performs the driving action for the underlying vehicle implementation.
10       * @return an action verb describing the motion (e.g., "drives", "pedals").
11       */
12      String drive();
13  
14      /**
15       * Default starting behaviour for drivable vehicles.
16       * Implementations may override to refine the start sequence.
17       * @return a small description of the start action.
18       */
19      default String start() {
20          return "start engine"; // default; bikes may ignore
21      }
22  }
