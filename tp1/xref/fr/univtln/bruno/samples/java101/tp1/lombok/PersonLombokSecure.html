3   import lombok.*;
4   
5   import java.util.Objects;
6   
7   /**
8    * Secure Lombok-backed Person: validation is performed at the factory/builder boundary.
9    *
10   * <p>This class demonstrates a recommended pattern when using Lombok:
11   * the public, validated static factory method {@link #of(String, String, int)} is
12   * annotated with {@code @Builder}. Lombok generates a builder whose {@code build()}
13   * method delegates to this factory method; therefore both calls to
14   * {@code PersonLombokSecure.of(...)} and to the generated builder
15   * {@code PersonLombokSecure.builder()...build()} use the same validation
16   * and normalization logic implemented in {@code of(...)}.</p>
17   *
18   * <p>Implementation notes:
19   * <ul>
20   *   <li>The all-args constructor is private (generated by Lombok with
21   *       {@code @AllArgsConstructor(access = AccessLevel.PRIVATE)}), ensuring
22   *       instances can only be created through the validated factory or the
23   *       generated builder.</li>
24   *   <li>The factory performs lightweight validation (null checks, range checks)
25   *       and normalization (e.g. {@code String.strip()}) before creating the
26   *       instance. Validation uses {@link Objects#requireNonNull} and throws
27   *       {@link IllegalArgumentException} for invalid numeric values.</li>
28   *   <li>We intentionally avoid using {@code @Data} here to keep the example
29   *       explicit; use {@code @Data} only when you want the convenience of
30   *       generating getters/setters and other boilerplate in one go.</li>
31   * </ul>
32   */
33  @AllArgsConstructor(access = AccessLevel.PRIVATE)
34  @Getter
35  @ToString
36  public class PersonLombokSecure {
37      private final String firstName;
38      private final String lastName;
39      @Setter
40      private int age;
41  
42      /**
43       * Private no-arg constructor kept for tools and to avoid a missing Javadoc
44       * warning for the implicit default constructor. Instances should be created
45       * through the validating factory {@link #of(String,String,int)} or the
46       * Lombok-generated builder.
47       */
48      @SuppressWarnings("unused")
49      private PersonLombokSecure() {
50          this.firstName = "";
51          this.lastName = "";
52          this.age = 0;
53      }
54  
55      @Builder
56      public static PersonLombokSecure of(String firstName, String lastName, int age) {
57          Objects.requireNonNull(firstName, "firstName must not be null");
58          Objects.requireNonNull(lastName, "lastName must not be null");
59          if (age < 0) throw new IllegalArgumentException("age must be >= 0");
60          return new PersonLombokSecure(firstName.strip(), lastName.strip(), age);
61      }
62  }
