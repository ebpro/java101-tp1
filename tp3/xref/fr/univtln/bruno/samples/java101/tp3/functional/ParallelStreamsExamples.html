10  /**
11   * Examples demonstrating parallel streams usage and common pitfalls.
12   *
13   * <p>Shows performance measurement and the dangers of shared mutable state when using
14   * parallel streams; recommends reduction operations as the correct pattern.</p>
15   */
16  @Slf4j
17  public class ParallelStreamsExamples {
18    /**
19     * Compare sequential vs parallel stream performance for a simple summation.
20     * This is illustrative only — real benchmarking requires JMH and controlling for warmup.
21     */
22    public static void parallelSumExample() {
23      log.info("=== Parallel sum example (measure) ===");
24      List<Integer> data = IntStream.rangeClosed(1, 1_000_000).boxed().toList();
25  
26      long t0 = System.currentTimeMillis();
27      long sumSeq = data.stream().mapToLong(Integer::longValue).sum();
28      long t1 = System.currentTimeMillis();
29  
30      log.debug("Sequential sum took {} ms", (t1 - t0));
31  
32      long sumParStart = System.currentTimeMillis();
33      long sumPar = data.parallelStream().mapToLong(Integer::longValue).sum();
34      long sumParEnd = System.currentTimeMillis();
35  
36      log.info("seq sum={} time={}ms", sumSeq, (t1 - t0));
37      log.info("par sum={} time={}ms", sumPar, (sumParEnd - sumParStart));
38      log.debug("Parallel sum took {} ms", (sumParEnd - sumParStart));
39    }
40  
41    /**
42     * Demonstrates the pitfall of mutating shared state (AtomicInteger used here still risks
43     * producing confusing results and should be avoided). Shows the correct pattern using reduction.
44     */
45    public static void sharedMutablePitfall() {
46      log.info("=== Shared mutable state pitfall ===");
47      List<Integer> data = IntStream.rangeClosed(1, 1000).boxed().toList();
48  
49      AtomicInteger counter = new AtomicInteger(0);
50      // WRONG: side-effect in parallel stream leads to race / unpredictable result
51      data.parallelStream().forEach(counter::addAndGet);
52      log.info("Counter (with parallel forEach) = {} (should be {})", counter.get(), data.stream().mapToInt(Integer::intValue).sum());
53  
54      int reduced = data.parallelStream().mapToInt(Integer::intValue).reduce(0, Integer::sum);
55      log.info("Reduced (parallel) = {}", reduced);
56  
57      // Student note: prefer stateless reduction or collectors. For example, using a thread-safe collector:
58      // int sum = data.parallelStream().mapToInt(Integer::intValue).sum(); // built-in is safe
59      // Or: data.parallelStream().collect(Collectors.summingInt(Integer::intValue));
60    }
61  
62    /**
63     * Runner for parallel streams examples.
64     *
65     * @param args ignored
66     */
67    public static void main(String[] args) {
68      parallelSumExample();
69      sharedMutablePitfall();
70    }
71  }
