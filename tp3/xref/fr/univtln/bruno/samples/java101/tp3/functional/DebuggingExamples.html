7   /**
8    * Small helpers showing how to debug stream pipelines: peek, sampling and safe terminal operations.
9    */
10  @Slf4j
11  public class DebuggingExamples {
12    /**
13     * Private constructor to prevent instantiation.
14     */
15    private DebuggingExamples() {
16    }
17  
18    /**
19     * Demonstrate using peek() to inspect intermediate pipeline values.
20     */
21    public static void peekPipelineExample() {
22      log.info("[functional] peek example");
23      List<Integer> nums = List.of(1, 2, 3, 4, 5);
24      List<Integer> result = nums.stream()
25        .peek(n -> log.info("orig {}", n))
26        .filter(n -> n % 2 == 0)
27        .peek(n -> log.info("filtered {}", n))
28        .map(n -> n * 2)
29        .peek(n -> log.info("mapped {}", n))
30        .toList();
31      log.info("[functional] final result {}", result);
32    }
33  
34    /**
35     * Demonstrate safe usage of findFirst with ifPresentOrElse.
36     */
37    public static void safeFirstExample() {
38      log.info("[functional] safe first example");
39      List<Integer> empty = List.of();
40      empty.stream().findFirst().ifPresentOrElse(
41        v -> log.info("first {}", v),
42        () -> log.info("no elements")
43      );
44    }
45  }
