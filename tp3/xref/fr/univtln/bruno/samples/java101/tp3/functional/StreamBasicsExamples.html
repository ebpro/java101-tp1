24    /**
25     * Demonstrates various ways to create streams from collections and primitives.
26     */
27    public static void streamCreationExample() {
28      log.info("=== Stream Creation ===");
29      List<String> fromList = List.of("A", "B", "C");
30      log.debug("fromList snapshot: {}", fromList);
31      log.info("From list: {}", fromList.stream().toList());
32      log.info("From array: {}", Arrays.stream(new String[]{"X", "Y"}).toList());
33      log.info("From values: {}", Stream.of("1", "2", "3").toList());
34      log.info("Empty count: {}", Stream.empty().count());
35      log.info("Iterate limit: {}", Stream.iterate(0, n -> n + 1).limit(5).toList());
36      log.info("Generate random: {}", Stream.generate(Math::random).limit(3).toList());
37      log.info("Range: {}", IntStream.range(1, 5).boxed().toList());
38      log.info("RangeClosed: {}", IntStream.rangeClosed(1, 5).boxed().toList());
39  
40      // Student note: infinite streams (Stream.generate / Stream.iterate without limit) can exhaust memory
41      // or run indefinitely if not limited. Always apply limit() or takeWhile() when experimenting.
42    }

44    /**
45     * Demonstrates filtering a collection using stream filters.
46     */
47    public static void filterExample() {
48      log.info("=== Filter Example ===");
49      List<Person> people = List.of(
50        Person.of("Alice", "Smith", 30),
51        Person.of("Bob", "Jones", 25),
52        Person.of("Charlie", "Brown", 35),
53        Person.of("David", "Wilson", 28)
54      );
55      log.debug("people snapshot: {}", people);
56      List<String> age30 = people.stream().filter(p -> p.age() >= 30).map(Person::getFullName).toList();
57      log.debug("Filtered age>=30 count: {}", age30.size());
58      log.info("Age >=30: {}", age30);
59      List<String> complex = people.stream().filter(p -> p.age() >= 25).filter(p -> p.lastName().length() > 5).map(Person::getFullName).toList();
60      log.debug("Filtered complex pipeline count: {}", complex.size());
61      log.info("Age>=25 & long lastName: {}", complex);
62    }

64    /**
65     * Demonstrates mapping operations on streams (map, method references).
66     */
67    public static void mapExample() {
68      log.info("=== Map Example ===");
69      List<Person> people = List.of(Person.of("Alice", "Smith", 30), Person.of("Bob", "Jones", 25));
70      log.info("Full names: {}", people.stream().map(Person::getFullName).toList());
71      log.info("Ages: {}", people.stream().map(Person::age).toList());
72      log.info("Upper: {}", people.stream().map(Person::getFullName).map(String::toUpperCase).toList());
73    }

75    /**
76     * Demonstrates flatMap for nested collections and splitting strings.
77     */
78    public static void flatMapExample() {
79      log.info("=== FlatMap Example ===");
80      List<List<String>> nested = List.of(List.of("A", "B"), List.of("C"));
81      log.info("Flatten: {}", nested.stream().flatMap(Collection::stream).toList());
82      List<String> sentences = List.of("Hello World", "Java Streams");
83      log.info("Words: {}", sentences.stream().flatMap(s -> Arrays.stream(s.split(" "))).toList());
84    }

86    /**
87     * Demonstrates distinct/sorted/limit/skip stream terminal operations.
88     */
89    public static void distinctSortedLimitExample() {
90      log.info("=== Distinct/Sorted/Limit ===");
91      List<Integer> nums = List.of(5, 2, 8, 2, 9, 1, 5);
92      log.info("Distinct: {}", nums.stream().distinct().toList());
93      log.info("Sorted: {}", nums.stream().sorted().toList());
94      log.info("Top3 desc distinct: {}", nums.stream().distinct().sorted(Comparator.reverseOrder()).limit(3).toList());
95      log.info("Skip first 2 sorted: {}", nums.stream().sorted().skip(2).toList());
96    }

98    /**
99     * Demonstrates use of peek for debugging intermediate pipeline state.
100    */
101   public static void peekExample() {
102     log.info("=== Peek Example ===");
103     List<Integer> nums = List.of(1, 2, 3, 4, 5);
104     log.debug("nums initial: {}", nums);
105     List<Integer> result = nums.stream()
106       .peek(n -> log.debug("Original {}", n))
107       .filter(n -> n % 2 == 0)
108       .peek(n -> log.debug("Filtered {}", n))
109       .map(n -> n * 2)
110       .peek(n -> log.debug("Mapped {}", n))
111       .toList();
112     log.info("Result {}", result);
113 
114     // Student note: Stream.toList() (since Java 16) may return an unmodifiable list; Collectors.toList()
115     // historically returned a mutable list. If code expects mutability, use new ArrayList<>(stream.toList())
116     // or Collectors.toCollection(ArrayList::new).
117   }

119   /**
120    * Demonstrates different sorting idioms using streams and comparators.
121    */
122   public static void sortingExample() {
123     log.info("=== Sorting Example ===");
124     List<Person> people = List.of(Person.of("Charlie", "Brown", 35), Person.of("Alice", "Smith", 30), Person.of("Bob", "Jones", 25));
125     log.info("Natural: {}", people.stream().sorted().map(Person::getFullName).toList());
126     log.info("By age: {}", people.stream().sorted(Comparator.comparing(Person::age)).map(p -> p.getFullName() + "(" + p.age() + ")").toList());
127   }
