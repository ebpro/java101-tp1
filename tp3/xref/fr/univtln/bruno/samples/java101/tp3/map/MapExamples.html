22    /**
23     * Basic HashMap usage and update semantics.
24     */
25    public static void hashMapExample() {
26      log.info("=== HashMap Example ===");
27      Map<String, Person> people = new HashMap<>();
28      people.put("P001", Person.of("Alice", "Smith", 30));
29      people.put("P002", Person.of("Bob", "Jones", 25));
30      people.put("P003", Person.of("Charlie", "Brown", 35));
31      log.debug("Initial people map: {}", people);
32      log.info("Size: {}", people.size());
33      log.info("P001 -> {}", people.get("P001").getFullName());
34      Person prev = people.put("P001", Person.of("Alice", "Smith", 31));
35      log.debug("After updating P001 previous: {} current: {}", prev, people.get("P001"));
36      log.info("Updated P001 age {} -> {}", prev.age(), people.get("P001").age());
37      people.remove("P002");
38      log.debug("After remove P002 map: {}", people);
39      log.info("After remove P002 size: {}", people.size());
40    }

68    /**
69     * Modern Map methods: getOrDefault, putIfAbsent, computeIfAbsent, computeIfPresent, compute, merge.
70     */
71    public static void modernMapMethodsExample() {
72      log.info("=== Modern Map Methods ===");
73      Map<String, Integer> scores = new HashMap<>();
74      scores.put("Alice", 100);
75      scores.put("Bob", 85);
76      log.debug("Scores initial: {}", scores);
77      log.info("getOrDefault Charlie: {}", scores.getOrDefault("Charlie", 0));
78      scores.putIfAbsent("Bob", 90);
79      scores.putIfAbsent("David", 90);
80      scores.computeIfAbsent("Eve", k -> k.length() * 10);
81      scores.computeIfPresent("Alice", (k, v) -> v + 10);
82      scores.compute("Bob", (k, v) -> v == null ? 0 : v + 5);
83      scores.merge("Alice", 20, Integer::sum);
84      scores.merge("Frank", 75, Integer::sum);
85      scores.replace("Bob", 95);
86      log.debug("Scores after modifications: {}", scores);
87      log.info("Scores: {}", scores);
88  
89      // Student note: in multi-threaded contexts computeIfAbsent may race if the mapping function has side-effects.
90      // Prefer using ConcurrentHashMap for concurrent scenarios or make the mapping function side-effect free.
91      // Example (unsafe in multi-threaded code):
92      // map.computeIfAbsent(key, k -> heavyInit(k)); // heavyInit should not have side-effects visible outside
93    }

96     * Different ways to iterate a map's entries, keys and values.
97     */
98    public static void iterationExample() {
99      log.info("=== Iteration Example ===");
100     Map<String, Integer> ages = Map.of("Alice", 30, "Bob", 25, "Charlie", 35);
101     for (Map.Entry<String, Integer> e : ages.entrySet()) log.info("Entry {}={} ", e.getKey(), e.getValue());
102     ages.keySet().forEach(k -> log.info("Key {}", k));
103     ages.values().forEach(v -> log.info("Val {}", v));
104   }

106   /**
107    * Counting and frequency aggregation examples using loops and merge.
108    */
109   public static void countingExample() {
110     log.info("=== Counting Example ===");
111     List<String> words = List.of("apple", "banana", "apple", "cherry", "banana", "apple");
112     Map<String, Integer> manual = new HashMap<>();
113     for (String w : words) manual.put(w, manual.getOrDefault(w, 0) + 1);
114     Map<String, Integer> mergeMap = new HashMap<>();
115     words.forEach(w -> mergeMap.merge(w, 1, Integer::sum));
116     log.debug("Words list: {}", words);
117     log.info("Manual {} Merge {}", manual, mergeMap);
118   }

134   /**
135    * Immutable map creation and views.
136    */
137   public static void immutableMapExample() {
138     log.info("=== Immutable Map Example ===");
139     Map<String, Integer> m1 = Map.of("A", 1, "B", 2, "C", 3);
140     log.info("Map.of {}", m1);
141     try {
142       m1.put("D", 4);
143     } catch (UnsupportedOperationException e) {
144       log.info("Cannot modify immutable map");
145     }
146     Map<String, Integer> backing = new HashMap<>(Map.of("X", 1, "Y", 2));
147     Map<String, Integer> view = Collections.unmodifiableMap(backing);
148     backing.put("Z", 3);
149     log.info("View after backing change: {}", view);
150   }
