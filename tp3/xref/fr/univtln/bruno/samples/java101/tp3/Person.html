3   import lombok.With;
4   import lombok.NonNull;
5   
6   import java.util.Comparator;
7   import java.util.Objects;
8   
9   /**
10   * Immutable value object representing a person used in TP3 collection examples.
11   *
12   * <p>Instances are created via the static factory {@link #of(String, String, int)} which
13   * performs basic validation. The class implements {@link Comparable} to provide a
14   * natural ordering (lastName, firstName, age).</p>
15   *
16   * <p>Pedagogical note on <code>@With</code> and Java records:
17   * - Lombok's <code>@With</code> is applied to record components to generate immutable
18   *   "withX(...)" methods that are convenient in exercises where you want to
19   *   create a slightly modified copy (for example <code>p.withAge(31)</code>).
20   * - Java records already provide <code>equals</code>, <code>hashCode</code> and
21   *   <code>toString</code>. Lombok is used only to generate helper methods such as
22   *   the <code>with</code> methods. Ensure annotation processing is enabled in the IDE
23   *   or build so generated methods are visible.
24   *
25   * Important edge cases to illustrate in class:
26   * - Null handling: prefer factories (e.g. <code>of</code>) that validate parameters
27   *   instead of silently accepting null fields.
28   * - equals / compareTo consistency: ensure that <code>compareTo(a,b)==0</code>
29   *   implies <code>a.equals(b)</code> to avoid surprising behavior in sorted
30   *   collections (e.g. <code>TreeSet</code> / <code>TreeMap</code>).
31   * - Concurrent modification: modifying a collection while iterating can throw
32   *   <code>ConcurrentModificationException</code>. Demonstrate safe patterns such as
33   *   using an explicit <code>Iterator</code> and <code>iterator.remove()</code>, or
34   *   using concurrent collections like <code>CopyOnWriteArrayList</code> when appropriate.
35   * </p>
36   */
37  public record Person(@With String firstName, @With String lastName, @With int age) implements Comparable<Person> {
38    /**
39     * Create a new {@link Person} instance after validating arguments.
40     *
41     * @param firstName non-null, non-blank first name
42     * @param lastName  non-null, non-blank last name
43     * @param age       non-negative age
44     * @return a new immutable Person
45     * @throws IllegalArgumentException if any parameter is invalid
46     */
47    public static Person of(String firstName, String lastName, int age) {
48      Objects.requireNonNull(firstName, "First name cannot be null");
49      Objects.requireNonNull(lastName, "Last name cannot be null");
50      String fn = firstName.trim();
51      String ln = lastName.trim();
52      if (fn.isBlank()) throw new IllegalArgumentException("First name cannot be blank");
53      if (ln.isBlank()) throw new IllegalArgumentException("Last name cannot be blank");
54      if (age < 0) throw new IllegalArgumentException("Age cannot be negative");
55      return new Person(fn, ln, age);
56    }
57  
58    /**
59     * Natural ordering: by last name, then first name, then age.
60     *
61     * @param other the other person to compare with
62     * @return negative/zero/positive as this is less/equal/greater than other
63     */
64    @Override
65    public int compareTo(@NonNull Person other) {
66      // Comparable contract expects non-null argument; Lombok @NonNull documents and enforces this
67      return Comparator
68        .comparing(Person::lastName)
69        .thenComparing(Person::firstName)
70        .thenComparingInt(Person::age)
71        .compare(this, other);
72    }
73  
74    /**
75     * Return the human-friendly full name combining first and last name.
76     *
77     * @return full name ("First Last")
78     */
79    public String getFullName() {
80      return firstName + " " + lastName;
81    }
82  
83    /**
84     * Small demo for students showing the Lombok-generated `with` methods on record components.
85     * This main method is intentionally minimal: it creates a Person, uses the generated
86     * withAge / withFirstName methods to create modified copies, and prints results so
87     * students can see the immutability behavior.
88     */
89    public static void main(String[] args) {
90      Person p = Person.of("Alice", "Smith", 30);
91      System.out.println("Original: " + p);
92  
93      // Lombok generates withAge(int) and withFirstName(String) because of @With on components
94      Person older = p.withAge(31);
95      System.out.println("After withAge(31): " + older);
96  
97      Person renamed = p.withFirstName("Alicia");
98      System.out.println("After withFirstName(\"Alicia\"): " + renamed);
99  
100     // original instance remains unchanged (immutability)
101     System.out.println("Original still: " + p);
102   }
103 }
