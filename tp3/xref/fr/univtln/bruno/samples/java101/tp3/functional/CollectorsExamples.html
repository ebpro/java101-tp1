24    /**
25     * Showcases terminal stream operations like count/any/all/none/findFirst.
26     */
27    public static void terminalOperationsExample() {
28      log.info("[functional] Terminal operations: count/any/all/none/findFirst");
29      List<Integer> nums = List.of(1, 2, 3, 4, 5, 6);
30      long evenCount = nums.stream().filter(n -> n % 2 == 0).count();
31      boolean anyGt4 = nums.stream().anyMatch(n -> n > 4);
32      boolean allPositive = nums.stream().allMatch(n -> n > 0);
33      boolean noneNegative = nums.stream().noneMatch(n -> n < 0);
34      Optional<Integer> firstGt3 = nums.stream().filter(n -> n > 3).findFirst();
35      log.debug("nums: {}", nums);
36      log.info("evenCount={} anyGt4={} allPositive={} noneNegative={} firstGt3={}", evenCount, anyGt4, allPositive, noneNegative, firstGt3.orElse(-1));
37    }
38  
39    /**
40     * Demonstrates groupingBy and counting collectors on Person collections.
41     */
42    public static void groupingAndCountingExample() {
43      log.info("[functional] groupingBy and counting");
44      List<Person> people = List.of(Person.of("Alice", "Smith", 30), Person.of("Bob", "Jones", 25), Person.of("Charlie", "Brown", 35), Person.of("David", "Smith", 28));
45      log.debug("people for grouping: {}", people);
46      Map<String, List<Person>> byLast = people.stream().collect(Collectors.groupingBy(Person::lastName));
47      byLast.forEach((ln, ps) -> log.info("{} -> {}", ln, ps.stream().map(Person::getFullName).toList()));
48      Map<String, Long> counts = people.stream().collect(Collectors.groupingBy(Person::lastName, Collectors.counting()));
49      log.debug("group counts: {}", counts);
50      log.info("counts: {}", counts);
51    }

53    /**
54     * Demonstrates complex collectors (mapping + grouping) using Book examples.
55     */
56    public static void complexCollectorsBooksExample() {
57      log.info("[functional] complex collectors for Book examples");
58      List<Book> books = List.of(
59        Book.of("978-0134685991", "Effective Java", "Joshua Bloch", LocalDate.of(2018, 1, 6), 45.99),
60        Book.of("978-0596009205", "Head First Java", "Kathy Sierra", LocalDate.of(2005, 2, 9), 39.99),
61        Book.of("978-0134685992", "Java Concurrency", "Brian Goetz", LocalDate.of(2006, 5, 19), 42.99)
62      );
63      log.debug("books: {}", books);
64      Map<String, List<String>> titlesByAuthor = books.stream().collect(Collectors.groupingBy(Book::author, Collectors.mapping(Book::title, Collectors.toList())));
65      log.debug("titlesByAuthor: {}", titlesByAuthor);
66      titlesByAuthor.forEach((a, t) -> log.info("{} -> {}", a, t));
67  
68      // Student note: beware of Collectors.toMap when keys might collide. The 2-arg toMap throws
69      // IllegalStateException on duplicate keys. Use the 3-arg version to resolve duplicates, for example:

71      // Or groupingBy when keys are not unique.
72    }
