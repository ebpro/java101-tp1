3   import lombok.With;
4   
5   import java.time.LocalDate;
6   import java.util.Comparator;
7   
8   /**
9    * Immutable value object representing a book used in TP3 sorting and grouping examples.
10   *
11   * <p>Provides a static factory {@link #of(String, String, String, LocalDate, double)} which validates inputs.
12   * Equality and hash code are based on all fields (generated by the Java record implementation).
13   * Natural ordering is implemented in a functional style using a Comparator chain and is consistent with equals()/hashCode().</p>
14   *
15   * <p>Note pédagogique sur <code>@With</code> :
16   * - Le code utilise Lombok's <code>@With</code> annotation sur les components du <code>record</code> pour générer
17   *   des méthodes immuables de type <code>withX(...)</code> (p. ex. <code>book.withPrice(12.0)</code>), facilitant
18   *   la modification immuable d'instances.
19   * - Pour que Lombok génère ces méthodes, l'annotation processing doit être activée dans l'IDE (ou lors de la compilation).
20   * - Alternative sans Lombok : écrire explicitement une méthode de copie (p. ex. <code>withPrice</code>) qui retourne
21   *   une nouvelle instance du <code>record</code> avec le champ modifié.
22   * - Note : les records Java fournissent déjà equals/hashCode/toString automatiquement ; Lombok n'est pas nécessaire
23   *   pour ces comportements mais propose des utilitaires pratiques comme <code>@With</code>.</p>
24   */
25  public record Book(@With String isbn, @With String title, @With String author, @With LocalDate publishedDate,
26                     @With double price) implements Comparable<Book> {
27    /**
28     * Create a validated Book instance.
29     *
30     * @param isbn          non-null, non-blank ISBN
31     * @param title         non-null, non-blank title
32     * @param author        non-null, non-blank author
33     * @param publishedDate non-null publication date
34     * @param price         non-negative price
35     * @return a new Book instance
36     */
37    public static Book of(String isbn, String title, String author, LocalDate publishedDate, double price) {
38      if (isbn == null || isbn.isBlank()) throw new IllegalArgumentException("ISBN cannot be null or blank");
39      if (title == null || title.isBlank()) throw new IllegalArgumentException("Title cannot be null or blank");
40      if (author == null || author.isBlank()) throw new IllegalArgumentException("Author cannot be null or blank");
41      if (publishedDate == null) throw new IllegalArgumentException("Published date cannot be null");
42      if (price < 0) throw new IllegalArgumentException("Price cannot be negative");
43      return new Book(isbn, title, author, publishedDate, price);
44    }
45  
46    /**
47     * Natural ordering implemented in a functional style: compare by ISBN, then title, then author,
48     * then published date, then price. Implemented using a Comparator chain for clarity and
49     * conciseness. The ordering is chosen to be consistent with equals()/hashCode() produced by
50     * the Java record implementation to avoid surprises when Books are used in sorted collections.
51     */
52    @Override
53    public int compareTo(Book other) {
54      if (other == null) return 1;
55      return Comparator
56        .comparing(Book::isbn, Comparator.nullsFirst(String::compareTo))
57        .thenComparing(Book::title, Comparator.nullsFirst(String::compareTo))
58        .thenComparing(Book::author, Comparator.nullsFirst(String::compareTo))
59        .thenComparing(Book::publishedDate, Comparator.nullsFirst(LocalDate::compareTo))
60        .thenComparingDouble(Book::price)
61        .compare(this, other);
62    }
63  
64    /**
65     * Return a short description of the book (title, author and price).
66     *
67     * @return human readable description
68     */
69    public String getDescription() {
70      return String.format("%s by %s (%.2f€)", title, author, price);
71    }
72  
73    /**
74     * Small demo showing the Lombok-generated `with` methods on Book record components.
75     * It illustrates creating a Book, then creating modified copies using withPrice and withTitle.
76     */
77    public static void main(String[] args) {
78      Book b = Book.of("978-0134685991", "Effective Java", "Joshua Bloch", LocalDate.of(2018, 1, 6), 45.99);
79      System.out.println("Original: " + b);
80  
81      Book cheaper = b.withPrice(39.99);
82      System.out.println("After withPrice(39.99): " + cheaper);
83  
84      Book renamed = b.withTitle("Effective Java - 3rd Edition");
85      System.out.println("After withTitle(...): " + renamed);
86  
87      // original instance remains unchanged
88      System.out.println("Original still: " + b);
89    }
90  }
