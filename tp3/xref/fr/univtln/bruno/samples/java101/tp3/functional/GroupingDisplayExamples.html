10  /**
11   * Examples demonstrating groupingBy / partitioningBy and how to display grouped results.
12   */
13  @Slf4j
14  public class GroupingDisplayExamples {
15    /**
16     * Private constructor to prevent instantiation.
17     */
18    private GroupingDisplayExamples() {
19    }
20  
21    /**
22     * Display grouping by last name.
23     *
24     * @param people list of persons to group
25     */
26    public static void showGroupingByLastName(List<Person> people) {
27      Map<String, List<Person>> byLast = people.stream().collect(Collectors.groupingBy(Person::lastName));
28      byLast.forEach((ln, ps) -> log.info("[functional] {} -> {}", ln, ps.stream().map(Person::getFullName).toList()));
29    }
30  
31    /**
32     * Display partitioning of persons by age threshold.
33     *
34     * @param people    list of persons to partition
35     * @param threshold inclusive age threshold for the partition
36     */
37    public static void showPartitionByAge(List<Person> people, int threshold) {
38      var part = people.stream().collect(Collectors.partitioningBy(p -> p.age() >= threshold));
39      log.info("[functional] >={} {} <{} {}", threshold, part.get(true).stream().map(Person::getFullName).toList(), threshold, part.get(false).stream().map(Person::getFullName).toList());
40    }
41  }
