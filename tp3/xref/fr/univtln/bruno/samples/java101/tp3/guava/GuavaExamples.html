7   /**
8    * Small Guava Multimap example used for demonstration in TP3.
9    * A multimap allows multiple values per key.
10   *
11   * Student notes:
12   * - Guava Multimap is convenient when you need multiple values per key without managing lists yourself.
13   * - Compared to Map<K, List<V>> it provides cleaner semantics and utility methods.
14   * - If Guava isn't available, consider using Map<K, List<V>> with computeIfAbsent.
15   */
16  @Slf4j
17  public class GuavaExamples {
18    /**
19     * Private constructor to prevent instantiation.
20     */
21    private GuavaExamples() {
22    }
23  
24    /**
25     * Small Guava Multimap example used for demonstration in TP3.
26     *
27     * @param args ignored
28     */
29    public static void main(String[] args) {
30      Multimap<String, String> mm = ArrayListMultimap.create();
31      mm.put("key1", "v1");
32      mm.put("key1", "v2");
33      mm.put("key2", "v3");
34  
35      for (String k : mm.keySet()) {
36        log.info("{} -> {}", k, mm.get(k));
37      }
38  
39      // Student note (commented): equivalent using Map<K, List<V> > with computeIfAbsent
40      // Map<String, List<String>> map = new HashMap<>();
41      // map.computeIfAbsent("key1", k -> new ArrayList<>()).add("v1");
42      // map.computeIfAbsent("key1", k -> new ArrayList<>()).add("v2");
43    }
44  }
