11  @Slf4j
12  public final class DedupExamples {
13    // Prevent instantiation

15    /**
16     * Private constructor for utility class.

18    private DedupExamples() {
19    }

21    /**
22     * Show deduplication using stream distinct(). This is the preferred functional style when you already have a stream-based pipeline.
23     *
24     * @param withDup list containing duplicated elements
25     */
26    public static void dedupWithStreamDistinct(List<String> withDup) {
27      log.info("[functional] Unique (Stream.distinct): {}", withDup.stream().distinct().toList());
28    }
29  
30    /**
31     * Show collecting into a LinkedHashSet via stream then LinkedHashSet constructor.
32     *
33     * @param withDup list containing duplicated elements
34     */
35    public static void collectToLinkedHashSet(List<String> withDup) {
36      var ordered = new LinkedHashSet<>(withDup.stream().distinct().toList());
37      log.info("[functional] Collected LinkedHashSet (from stream): {}", ordered);
38    }
39  }
